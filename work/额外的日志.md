# 任务需求 添加三方日志

1, 将下面的表添加 到 app/db/models
-- ----------------------------
-- Table structure for system_logs
-- ----------------------------
DROP TABLE IF EXISTS `system_logs`;
CREATE TABLE `system_logs`  (
  `id` bigint unsigned NOT NULL COMMENT '主键ID',
  `log_time` datetime(3) NOT NULL COMMENT '日志时间(精确到毫秒)',
  `log_type` tinyint unsigned NOT NULL COMMENT '日志类型(1:DEBUG 2:INFO 3:SUCCESS 4:WARNING 5:ERROR)',
  `info_type` tinyint unsigned NOT NULL COMMENT '信息类型(1:系统 2:数据库 3:设备 4:相机 5:图像处理 6:路径规划 7:指令执行 8:夹具控制)',
  `content` text CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '日志内容',
  PRIMARY KEY (`id`) USING BTREE,
  INDEX `idx_time`(`log_time`) USING BTREE,
  INDEX `idx_log_type`(`log_type`) USING BTREE,
  INDEX `idx_info_type`(`info_type`) USING BTREE,
  INDEX `idx_time_type`(`log_time`, `log_type`, `info_type`) USING BTREE
) ENGINE = InnoDB AUTO_INCREMENT = 1 CHARACTER SET = utf8mb4 COLLATE = utf8mb4_unicode_ci COMMENT = '系统日志表' ROW_FORMAT = Dynamic

2，在 app/controller/http_prod.py 启动时，记录时间，根据这个时间作为起点，检测 system_logs新添加的日志， 发送到 app/server/run_api.py 在界面中显示
