```c++
// 路径点位结构体
struct s_PPts
{
    double fX;              // X坐标
    double fY;              // Y坐标  
    double fZ;              // Z坐标
    int iDef;               // 点属性定义：1-起点，2-终点，3-中间点，4-空跑点
    double fRow;            // 图像y坐标
    double fCol;            // 图像x坐标
    std::string strQgNotSafe; // 避让气缸索引，例如："0,2,4" 代表第1、3、5个气缸需要松开

    double ZMaxRelDm;       // 粒子相对于基准面的最大Z高度
    double MxHeightCur;     // 当前打磨深度
};

// 粒子路径规划结果参数
class s_LzppRtsPara
{
public:
    bool bTJG;                          // 综合判断结果
    double time;                        // 规划时间
    int iNumPts;                        // 点位数量
    std::vector<s_PPts> sListPPtsImage; // 图像坐标系下的点位列表
    std::vector<s_PPts> sListPPtsRobot; // 机床坐标系下的点位列表
    HObject RegionMxAllList;            // 规划后的图像加工区域


};
```
